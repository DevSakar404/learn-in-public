# Page Chat (F12) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** a toggleable chat sidebar on every blog post and playground page where readers highlight text or code and ask questions, answered by a LangGraph agent with the page as context.

**Architecture:** `useChat` (AI SDK) POSTs `{ slug, kind, selection, messages }` to a thin route. `ChatService` rate-limits, loads the published page (and lesson code) on the server, and builds a `PageContext` with a guarded `readRelated`. `LangGraphPageTutor` runs `createAgent` with one tool and streams through `@ai-sdk/langchain` into an AI SDK UI message stream `Response`.

**Tech Stack:** Next.js 16 route handler · `langchain@1.6` `createAgent` (LangGraph) · `@langchain/openai` · `ai@7` + `@ai-sdk/react@4` + `@ai-sdk/langchain@3` · Zod 4 · Vitest.

**Spec:** [docs/features/F12-page-chat.md](../../features/F12-page-chat.md)

## Global Constraints

- Layers: `domain ← application ← infrastructure ← lib (container) ← app ← components`. ESLint enforces it.
- Only `src/infrastructure/**` imports `langchain` and `@langchain/*` (already banned elsewhere in `eslint.config.mjs`).
- Services return `Result<T, DomainError>`; they never throw expected failures.
- No `any`. `strict` + `noUncheckedIndexedAccess`.
- The browser never sends page text. Only published notes reach the model.
- Limits: 10 messages per request, 1,000 chars per reader message, 4,000 per earlier answer, 2,000 chars of selection. 10 messages per IP per minute.
- Model: OpenAI `gpt-4o-mini`, `maxTokens: 500`, `timeout: 30_000`, `maxRetries: 0`, `recursionLimit: 7`.
- Reader-facing copy (exact): "That message is too long." · "You've asked a lot. Wait a minute and try again." · "Chat isn't available on this page." · "Chat isn't switched on yet." · "That note isn't available." · "Something went wrong. Try again."
- Checks after every task: `pnpm lint && pnpm typecheck && pnpm test`.
- Commits: conventional messages, each ending with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```ts
import type { GenerationError } from "@/domain/shared/errors";
import type { Result } from "@/domain/shared/result";

/** One chat message reduced to its text. Tool calls and other parts never reach the model as history. */
export interface ChatTurn {
  role: "user" | "assistant";
  text: string;
}

export interface RelatedNote {
  slug: string;
  title: string;
}

/** What the tutor knows about the page. Built on the server from published notes only. */
export interface PageContext {
  title: string;
  topic: string;
  /** The note's markdown without the playground fence. */
  markdown: string;
  /** The lesson code on playground pages, else null. */
  code: string | null;
  /** Text the reader highlighted, or "". */
  selection: string;
  /** Other published notes in the same topic. */
  related: RelatedNote[];
  /** A note from `related` as markdown; null for any other slug. */
  readRelated(slug: string): Promise<string | null>;
}

/** Answers a reader's question about a page. The Response body is an AI SDK UI message stream. */
export interface IPageTutor {
  answer(
    context: PageContext,
    turns: ChatTurn[],
    signal?: AbortSignal,
  ): Promise<Result<Response, GenerationError>>;
}
```

```ts
export class FakePageTutor implements IPageTutor {
  readonly calls: { context: PageContext; turns: ChatTurn[] }[] = [];
  failWith: GenerationError | null = null;

  async answer(
    context: PageContext,
    turns: ChatTurn[],
  ): Promise<Result<Response, GenerationError>> {
    this.calls.push({ context, turns });
    if (this.failWith) return err(this.failWith);
    return ok(new Response("fake answer"));
  }
}
```

```ts
import { z } from "zod";
import { lessonCode } from "@/domain/learn/lesson-code";
import type { ChatTurn, IPageTutor, PageContext } from "@/domain/learn/page-tutor";
import { splitPlayground } from "@/domain/note/lesson";
import type { INoteReader } from "@/domain/note/note-repository";
import { GenerationError, NotFoundError, type DomainError } from "@/domain/shared/errors";
import { err, type Result } from "@/domain/shared/result";
import type { RateLimiter } from "./rate-limiter";

const MAX_CHARS = { user: 1_000, assistant: 4_000 } as const;

/** An AI SDK UI message, reduced to its text. Other parts (tool calls, steps) are dropped. */
const messageSchema = z
  .object({
    role: z.enum(["user", "assistant"]),
    parts: z.array(z.looseObject({ type: z.string(), text: z.unknown() })).max(50),
  })
  .transform(({ role, parts }): ChatTurn => ({
    role,
    text: parts
      .flatMap((p) => (p.type === "text" && typeof p.text === "string" ? [p.text] : []))
      .join("\n"),
  }))
  .refine((turn) => turn.text.length <= MAX_CHARS[turn.role], "That message is too long.");

export const chatSchema = z.object({
  slug: z.string().min(1).max(200),
  kind: z.enum(["post", "playground"]),
  selection: z.string().max(2_000, "That selection is too long.").default(""),
  messages: z
    .array(messageSchema)
    .min(1)
    .max(10, "Too many messages.")
    .refine((turns) => {
      const last = turns.at(-1);
      return last?.role === "user" && last.text.trim() !== "";
    }, "Ask a question first."),
});
export type ChatInput = z.output<typeof chatSchema>;

/** Answers readers' questions about a published page. Every message is a paid model call, so each one is rate limited. */
export class ChatService {
  constructor(
    private readonly notes: INoteReader,
    private readonly tutor: IPageTutor,
    private readonly limiter: RateLimiter,
  ) {}

  async answer(
    clientKey: string,
    input: ChatInput,
    signal?: AbortSignal,
  ): Promise<Result<Response, DomainError>> {
    if (!this.limiter.take(clientKey)) {
      return err(
        new GenerationError("rate_limit", "You've asked a lot. Wait a minute and try again."),
      );
    }

    const found = await this.notes.findBySlug(input.slug);
    if (!found.ok || found.value.status !== "published") {
      return err(new NotFoundError("Note", input.slug));
    }
    const note = found.value;
    const { body, playground } = splitPlayground(note.contentMd);
    const code = playground ? lessonCode(playground) : null;
    if (input.kind === "playground" && code === null) {
      return err(new NotFoundError("Playground", input.slug));
    }

    const siblings = (
      await this.notes.list({ status: "published", topicId: note.topic.id, limit: 50 })
    ).filter((n) => n.id !== note.id);
    const bySlug = new Map(siblings.map((n) => [n.slug, n]));

    const context: PageContext = {
      title: note.title,
      topic: note.topic.name,
      markdown: body,
      code: input.kind === "playground" ? code : null,
      selection: input.selection,
      related: siblings.map((n) => ({ slug: n.slug, title: n.title })),
      readRelated: async (slug) => {
        const n = bySlug.get(slug);
        return n ? `# ${n.title}\n\n${splitPlayground(n.contentMd).body}` : null;
      },
    };
    return this.tutor.answer(context, input.messages, signal);
  }
}
```

```ts
async function collect<T>(stream: ReadableStream<T>): Promise<T[]> {
  const reader = stream.getReader();
  const out: T[] = [];

    if (done) return out;
```

```ts
import { toUIMessageStream } from "@ai-sdk/langchain";
import { ChatOpenAI } from "@langchain/openai";
import { createUIMessageStreamResponse, type UIMessageChunk } from "ai";
import { AIMessage, createAgent, HumanMessage, tool } from "langchain";
import { z } from "zod";
import type { ChatTurn, IPageTutor, PageContext } from "@/domain/learn/page-tutor";
import { GenerationError } from "@/domain/shared/errors";
import { err, ok, type Result } from "@/domain/shared/result";
import { toGenerationError } from "./error-mapping";

export const TUTOR_MODEL = "gpt-4o-mini";

export function systemPrompt(c: PageContext): string {
  return [
    "You are a friendly tutor on a learn-in-public blog about AI engineering.",
    `Answer questions about the page the reader is on: "${c.title}" (topic: ${c.topic}).`,
    "Stay on this page's topic. If a question is unrelated, say so in one sentence and steer back.",
    "Keep answers short and plain. Use markdown, and fenced code blocks for code.",
    "If the question needs another note from the list below, call readRelatedNote with its slug.",
    "",
    "<page>",
    c.markdown,
    "</page>",
    ...(c.code ? ["", "<code>", c.code, "</code>"] : []),
    ...(c.selection
      ? ["", "The reader highlighted:", "<selection>", c.selection, "</selection>"]
      : []),
    "",
    "Other notes in this topic:",
    ...(c.related.length > 0 ? c.related.map((n) => `- ${n.slug}: ${n.title}`) : ["(none)"]),
  ].join("\n");
}

/** toUIMessageStream sends error.message to the browser; swap it for a safe message. */
export function redactErrors(
  messageFor: () => string,
): TransformStream<UIMessageChunk, UIMessageChunk> {
  return new TransformStream({
    transform(chunk, controller) {
      controller.enqueue(
        chunk.type === "error" ? { type: "error", errorText: messageFor() } : chunk,
      );
    },
  });
}

/** Public page: keep provider details in the server log, not on screen. */
function publicMessage(error: GenerationError | null): string {
  if (error?.kind === "timeout") return error.message;
  if (error?.kind === "rate_limit") return "OpenAI is busy right now. Try again in a minute.";
  return "Something went wrong. Try again.";
}

/** The page chat agent: LangGraph's prebuilt agent loop with one tool, streamed to the AI SDK. */
export class LangGraphPageTutor implements IPageTutor {
  private model: ChatOpenAI | undefined;

  constructor(private readonly apiKey: string | undefined) {}

  async answer(
    context: PageContext,
    turns: ChatTurn[],
    signal?: AbortSignal,
  ): Promise<Result<Response, GenerationError>> {
    if (!this.apiKey) return err(new GenerationError("provider", "Chat isn't switched on yet."));
    this.model ??= new ChatOpenAI({
      model: TUTOR_MODEL,
      apiKey: this.apiKey,
      maxTokens: 500,
      timeout: 30_000,
      maxRetries: 0,
    });

    const readRelatedNote = tool(
      async ({ slug }) => (await context.readRelated(slug)) ?? "That note isn't available.",
      {
        name: "readRelatedNote",
        description:
          "Read another note in this topic by its slug. Use it only when the question needs that note.",
        schema: z.object({ slug: z.string() }),
      },
    );
    const agent = createAgent({
      model: this.model,
      tools: [readRelatedNote],
      systemPrompt: systemPrompt(context),
    });

    let failure: GenerationError | null = null;
    const logFailure = (error: unknown) => {
      failure = toGenerationError(error);
      console.error(`[ai] chat ${failure.kind}`, { model: TUTOR_MODEL, error });
    };
    try {
      const stream = await agent.stream(
        {
          messages: turns.map((t) =>
            t.role === "user" ? new HumanMessage(t.text) : new AIMessage(t.text),
          ),
        },
        // Each model call and each tool run is one step: 7 allows about 3 model calls.
        { streamMode: ["values", "messages"], recursionLimit: 7, signal },
      );
      const ui = toUIMessageStream(stream, { onError: logFailure }).pipeThrough(
        redactErrors(() => publicMessage(failure)),
      );
      return ok(createUIMessageStreamResponse({ stream: ui }));
    } catch (error) {
      // Fails before any streaming (bad request, network). Map it directly: TypeScript can't see
      // logFailure's write to `failure` here.
      const mapped = toGenerationError(error);
      console.error(`[ai] chat ${mapped.kind}`, { model: TUTOR_MODEL, error });
      return err(new GenerationError(mapped.kind, publicMessage(mapped)));
    }
  }
}
```

```ts
import { chatSchema } from "@/application/chat-service";
import { GenerationError, NotFoundError } from "@/domain/shared/errors";
import { container } from "@/lib/container";

// A tool call plus a streamed answer can outlast the platform's default limit.
export const maxDuration = 60;

/**
 * Public page chat. ChatService rate-limits by IP.
 * Errors are plain text: useChat shows the response body as the error message.
 */
export async function POST(request: Request) {
  const body: unknown = await request.json().catch(() => null);
  const input = chatSchema.safeParse(body);
  if (!input.success) {
    return new Response(input.error.issues[0]?.message ?? "That message is too long.", {
      status: 400,
    });
  }

  const ip = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "unknown";
  const result = await container.public.chat().answer(ip, input.data, request.signal);
  if (result.ok) return result.value;

  const error = result.error;
  if (error instanceof NotFoundError) {
    return new Response("Chat isn't available on this page.", { status: 404 });
  }
  const status = error instanceof GenerationError && error.kind === "rate_limit" ? 429 : 503;
  return new Response(error.message, { status });
}
```

```tsx
"use client";

import { MessageCircle } from "lucide-react";
import { useEffect, useState } from "react";
import { Button } from "@/components/ui/button";

interface Pick {
  text: string;
  top: number;
  left: number;
}

/** Shows an "Ask about this" button under text selected inside [data-chat-context]. */
export function SelectionAsk({ onAsk }: { onAsk: (text: string) => void }) {
  const [pick, setPick] = useState<Pick | null>(null);

  useEffect(() => {
    function update() {
      const selection = document.getSelection();
      const text = selection?.toString().trim() ?? "";
      const range = selection && selection.rangeCount > 0 ? selection.getRangeAt(0) : null;
      const node = range?.commonAncestorContainer;
      const element = node instanceof Element ? node : node?.parentElement;
      if (!text || !range || !element?.closest("[data-chat-context]")) return setPick(null);
      const rect = range.getBoundingClientRect();
      setPick({ text, top: rect.bottom + 8, left: Math.min(rect.left, window.innerWidth - 170) });
    }
    document.addEventListener("selectionchange", update);
    window.addEventListener("scroll", update, { passive: true });
    return () => {
      document.removeEventListener("selectionchange", update);
      window.removeEventListener("scroll", update);
    };
  }, []);

  if (!pick) return null;
  return (
    <Button
      type="button"
      size="sm"
      className="fixed z-50 shadow-md"
      style={{ top: pick.top, left: Math.max(8, pick.left) }}
      // Keep the selection: a mousedown here would clear it and unmount this button before the click.
      onMouseDown={(e) => e.preventDefault()}
      onClick={() => {
        onAsk(pick.text);
        setPick(null);
      }}
    >
      <MessageCircle aria-hidden /> Ask about this
    </Button>
  );
}
```

```tsx
"use client";

import { useChat } from "@ai-sdk/react";
import { DefaultChatTransport } from "ai";
import { ArrowUp, MessageCircle, RotateCcw, Square, X } from "lucide-react";
import { useEffect, useRef, useState, useSyncExternalStore } from "react";

import { Label } from "@/components/ui/label";

import { SelectionAsk } from "./selection-ask";

const STORAGE_KEY = "page-chat-open";
const MAX_SENT = 10;
const MAX_SELECTION = 2_000;
const STARTERS = [
  "Explain this page in simpler words",
  "What should I remember from this?",
  "Show me a tiny example",
];

// Open/closed per browser (a convenience, not data). useSyncExternalStore keeps the server
// render (closed) and the first client render in agreement, like slide-images.tsx.
const noSubscribe = () => () => {};
function readOpen(): boolean {
  try {
    return localStorage.getItem(STORAGE_KEY) === "1";
  } catch {
    return false;
  }
}

/** The "Ask about this page" sidebar: a toggle, the panel, and the highlight-to-ask button. */
export function PageChat({ slug, kind }: { slug: string; kind: "post" | "playground" }) {
  const stored = useSyncExternalStore(noSubscribe, readOpen, () => false);
  const [picked, setPicked] = useState<boolean | null>(null);
  const open = picked ?? stored;
  const [focus, setFocus] = useState<"input" | "toggle" | null>(null);
  const [selection, setSelection] = useState("");
  const [draft, setDraft] = useState("");
  const toggleRef = useRef<HTMLButtonElement>(null);
  const inputRef = useRef<HTMLTextAreaElement>(null);

  // The transport is built once; it reads the current selection when a request is sent.
  const selectionRef = useRef(selection);
  useEffect(() => {
    selectionRef.current = selection;
  }, [selection]);
  const [transport] = useState(
    () =>
      new DefaultChatTransport({
        api: "/api/chat",
        prepareSendMessagesRequest: ({ messages }) => ({
          body: {
            slug,
            kind,
            selection: selectionRef.current,
            messages: messages.slice(-MAX_SENT),
          },
        }),
      }),
  );
  const { messages, sendMessage, status, stop, error, regenerate, setMessages } = useChat({
    transport,
  });
  const busy = status === "submitted" || status === "streaming";

  // Move focus only when the reader opens or closes the panel, never on page load.
  useEffect(() => {
    if (focus === "input") inputRef.current?.focus();
    if (focus === "toggle") toggleRef.current?.focus();
  }, [focus]);

  function setOpen(next: boolean) {
    setPicked(next);
    setFocus(next ? "input" : "toggle");
    try {
      localStorage.setItem(STORAGE_KEY, next ? "1" : "0");
    } catch {
      // storage unavailable (private mode): the choice just isn't remembered
    }
  }

  function ask(text: string) {
    const question = text.trim();
    if (!question || busy) return;
    void sendMessage({ text: question });
    setDraft("");
  }

  return (
    <>
      <SelectionAsk
        onAsk={(text) => {
          setSelection(text.slice(0, MAX_SELECTION));
          setOpen(true);
        }}
      />
      <Button
        ref={toggleRef}
        type="button"
        size="icon"
        className={open ? "hidden" : "fixed right-4 bottom-4 z-40 size-12 rounded-full shadow-lg"}
        aria-label="Ask about this page"
        aria-expanded={open}
        aria-controls="page-chat"
        onClick={() => setOpen(true)}
      >
        <MessageCircle aria-hidden />
      </Button>
      <aside
        id="page-chat"
        aria-label="Ask about this page"
        hidden={!open}
        data-chat-open={open || undefined}
        onKeyDown={(e) => {
          if (e.key === "Escape") setOpen(false);
        }}
        className="fixed inset-0 z-50 flex flex-col bg-background lg:left-auto lg:w-[380px] lg:border-l"
      >
        <header className="flex items-center gap-2 border-b px-4 py-3">
          <h2 className="flex-1 font-semibold">Ask about this page</h2>
          <Button
            type="button"
            variant="ghost"
            size="sm"
            onClick={() => {
              void stop();
              setMessages([]);
              setSelection("");
            }}
          >
            New chat
          </Button>
          <Button
            type="button"
            variant="ghost"
            size="icon"
            aria-label="Close chat"
            onClick={() => setOpen(false)}
          >
            <X aria-hidden />
          </Button>
        </header>

        <div aria-live="polite" aria-busy={busy} className="flex-1 space-y-4 overflow-y-auto p-4">
          {messages.length === 0 && (
            <div className="space-y-2">
              <p className="text-sm text-muted-foreground">
                Highlight any text or code on the page, or start with:
              </p>
              {STARTERS.map((s) => (
                <Button
                  key={s}
                  type="button"
                  variant="outline"
                  size="sm"
                  className="h-auto w-full justify-start text-left whitespace-normal"
                  onClick={() => ask(s)}
                >
                  {s}
                </Button>
              ))}
            </div>
          )}
          {messages.map((m) => (
            <div
              key={m.id}
              className={
                m.role === "user" ? "ml-8 rounded-lg bg-muted px-3 py-2 text-sm" : "text-sm"
              }
            >
              {m.parts.map((p, i) => {
                if (p.type === "text") {
                  return m.role === "user" ? (
                    <p key={i} className="whitespace-pre-wrap">
                      {p.text}
                    </p>
                  ) : (
                    <Markdown key={i} className="prose-sm">
                      {p.text}
                    </Markdown>
                  );
                }
                if (p.type === "dynamic-tool" && p.toolName === "readRelatedNote") {
                  const slugRead =
                    typeof p.input === "object" && p.input !== null && "slug" in p.input
                      ? String(p.input.slug)
                      : "";
                  return (
                    <p key={i} className="text-xs text-muted-foreground italic">
                      Read note: {slugRead}
                    </p>
                  );
                }
                return null;
              })}
            </div>
          ))}
          {error && (
            <div role="alert" className="space-y-2 text-sm text-destructive">
              <p>{error.message || "Something went wrong. Try again."}</p>
              <Button type="button" variant="outline" size="sm" onClick={() => void regenerate()}>
                <RotateCcw aria-hidden /> Retry
              </Button>
            </div>
          )}
        </div>

        <form
          className="space-y-2 border-t p-3"
          onSubmit={(e) => {
            e.preventDefault();
            ask(draft);
          }}
        >
          {selection && (
            <div className="flex items-start gap-2 rounded border-l-2 border-primary bg-muted px-2 py-1 text-xs">
              <q className="line-clamp-3 flex-1 font-mono whitespace-pre-wrap">{selection}</q>
              <button
                type="button"
                aria-label="Remove highlighted text"
                className="rounded p-0.5 hover:bg-background"
                onClick={() => setSelection("")}
              >
                <X className="size-3" aria-hidden />
              </button>
            </div>
          )}
          <Label htmlFor="page-chat-input" className="sr-only">
            Your question
          </Label>
          <div className="flex items-end gap-2">
            <Textarea
              id="page-chat-input"
              ref={inputRef}
              rows={2}
              maxLength={1_000}
              value={draft}
              placeholder="Ask about this page"
              onChange={(e) => setDraft(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === "Enter" && !e.shiftKey && !e.nativeEvent.isComposing) {
                  e.preventDefault();
                  ask(draft);
                }
              }}
            />
            {busy ? (
              <Button type="button" size="icon" aria-label="Stop" onClick={() => void stop()}>
                <Square aria-hidden />
              </Button>
            ) : (
              <Button type="submit" size="icon" aria-label="Send" disabled={!draft.trim()}>
                <ArrowUp aria-hidden />
              </Button>
            )}
          </div>
        </form>
      </aside>
    </>
  );
}
```
